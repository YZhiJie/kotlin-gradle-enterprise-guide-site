D3 · 生产可观测性 ​

第二部「精通专题」第三篇。目标：让服务在出问题时能定位、能度量、能解释，而不是加几行 log.info 就宣布「可观测」。示例工程见 examples/backend-service。

D3.0 谁该读，解决什么问题 ​

读者典型痛点读完能做什么
后端工程师线上 P99 抖动，只有一堆零散日志用指标定位「哪个环节慢」，用 trace 定位「哪次请求慢」
SRE / 运维告警天天响，说不清服务到底健康不健康用 SLO + 错误预算替代拍脑袋阈值
架构师跨服务调用断链，trace 只能看到一半设计 W3C traceparent 的端到端传播
测试压测数据与线上指标对不上统一指标口径与标签口径

一句话：日志回答「发生了什么」，指标回答「整体怎么样」，追踪回答「这一次为什么慢」。三者缺一，排障就会退化成猜。

D3.2.2 Timer 与 suspend：为什么 recordCallable 是错的 ​

Timer.recordCallable 接收 java.util.concurrent.Callable，它不是 suspend 函数：在协程里写 checkoutTimer.recordCallable { delay(10) } 会直接编译失败。正确做法是「取 Sample → 执行业务 → finally 停止」：

examples/backend-service/src/main/kotlin/com/example/backend/ShopMetrics.kt可在线运行
kotlin
// 文件: ShopMetrics.kt —— 指标命名、标签基数治理与「能包住 suspend」的计时
// 配套章节：04-jvm-backend.md 4.7、mastery-observability.md D3.3
package com.example.backend

import io.micrometer.core.instrument.MeterRegistry
import io.micrometer.core.instrument.Timer
import org.springframework.stereotype.Component

/** 结果标签用枚举约束取值，避免把异常消息、用户 ID 等高基数字段写进标签。 */
enum class CallResult(val tag: String) { SUCCESS("success"), FAILURE("failure") }

/** 渠道标签同样有界：新增渠道需要改代码评审，而不是运行时随便传字符串。 */
enum class Channel(val tag: String) { WEB("web"), CLI("cli") }

@Component
class ShopMetrics(private val registry: MeterRegistry) {

    /**
     * 计时必须跨越挂起点，Micrometer 的 `recordCallable` 只接受同步 Callable，
     * 不能包 suspend 代码。正确做法是「先取 Sample，再在 finally 里 stop」。
     */
    suspend fun <T> timed(channel: Channel, block: suspend () -> T): T {
        val sample = Timer.start(registry)
        try {
            val value = block()
            registry.counter(
                "shop.orders.requests",
                "result", CallResult.SUCCESS.tag,
                "channel", channel.tag,
            ).increment()
            return value
        } catch (e: Throwable) {
            registry.counter(
                "shop.orders.requests",
                "result", CallResult.FAILURE.tag,
                "channel", channel.tag,
            ).increment()
            throw e
        } finally {
            sample.stop(
                Timer.builder("shop.orders.duration")
                    .tag("channel", channel.tag)
                    .publishPercentileHistogram()          // Prometheus 侧生成 _bucket，支持 P99 计算
                    .register(registry),
            )
        }
    }

    /** 记录一次可取消的下游超时，便于区分「超时」与「业务失败」。 */
    fun recordTimeout() {
        registry.counter("shop.orders.timeouts").increment()
    }
}

kotlin
// 文件: OrderService.kt —— 订单业务闭环：事务、幂等、分页、取消与可观测性
// 配套章节：04-jvm-backend.md 4.7 / 4.8、mastery-observability.md D3.2 / D3.3
package com.example.backend

import com.example.backend.order.CreateOrderCommand
import com.example.backend.order.Order
import com.example.backend.order.OrderNotFoundException
import com.example.backend.order.OrderPage
import com.example.backend.order.OrderRepository
import com.example.backend.order.OrderStateConflictException
import com.example.backend.order.OrderStatus
import com.example.backend.outbox.OrderEvent
import com.example.backend.outbox.OrderEventPublisher
import io.micrometer.tracing.Tracer
import org.springframework.dao.DuplicateKeyException
import org.springframework.stereotype.Service
import org.springframework.transaction.support.TransactionTemplate
import java.time.OffsetDateTime
import java.time.ZoneOffset

data class CreateOrderResult(val order: Order, val replayed: Boolean)

@Service
class OrderService(
    private val repository: OrderRepository,
    private val metrics: ShopMetrics,
    private val tracer: Tracer,
    private val transactionTemplate: TransactionTemplate,
    private val eventPublisher: OrderEventPublisher,
) {
    /**
     * 手动子 span 的正确姿势：span 是「对象」而不是「线程变量」，
     * 在挂起点前后传递的是同一个对象，所以可以安全地跨 suspend 使用。
     */
    suspend fun load(id: Long, channel: Channel): Order =
        observed("order.load", mapOf("order.id" to id.toString()), channel) {
            repository.findById(id) ?: throw OrderNotFoundException(id)
        }

    /**
     * 创建订单：以 Idempotency-Key 为幂等键。
     * 1. 先查同键记录，命中直接返回（重放）；
     * 2. 未命中则在同一事务内插入；
     * 3. 并发穿透时唯一约束抛 DuplicateKeyException，事务回滚后回读并返回同一条记录。
     */
    suspend fun create(command: CreateOrderCommand, channel: Channel): CreateOrderResult =
        observed("order.create", mapOf("idempotency.key" to command.idempotencyKey), channel) {
            repository.findByIdempotencyKey(command.idempotencyKey)?.let { existing ->
                return@observed CreateOrderResult(existing, replayed = true)
            }
            val id = try {
                transactionTemplate.execute {
                    val newId = repository.insert(command)
                    // D11：业务行与 outbox 行在同一事务提交——业务成功则事件必在，回滚则事件必不在
                    val now = OffsetDateTime.now(ZoneOffset.UTC)
                    eventPublisher.publish(
                        OrderEvent(
                            orderId = newId,
                            type = "ORDER_CREATED",
                            occurredAt = now,
                            payload = mapOf(
                                "customerId" to command.customerId,
                                "amountCents" to command.amountCents.toString(),
                            ),
                        ),
                    )
                    newId
                } ?: throw IllegalStateException("事务未返回主键")
            } catch (e: DuplicateKeyException) {
                // 另一个并发请求已经写入同一幂等键：事务已回滚，回读即可
                val existing = repository.findByIdempotencyKey(command.idempotencyKey)
                    ?: throw e
                return@observed CreateOrderResult(existing, replayed = true)
            }
            val created = repository.findById(id) ?: throw IllegalStateException("订单写入后不可见：$id")
            CreateOrderResult(created, replayed = false)
        }

    /** 取消订单：状态机 + 乐观并发（version 自增）；已取消返回 409 而不是静默成功。 */
    suspend fun cancel(id: Long, channel: Channel): Order =
        observed("order.cancel", mapOf("order.id" to id.toString()), channel) {
            val cancelled = transactionTemplate.execute {
                if (repository.cancel(id) == 0) {
                    val current = repository.findById(id) ?: throw OrderNotFoundException(id)
                    throw OrderStateConflictException(
                        if (current.status == OrderStatus.CANCELLED) "订单已取消：$id"
                        else "订单当前状态不允许取消：${current.status}",
                    )
                }
                val current = repository.findById(id) ?: throw OrderNotFoundException(id)
                // 取消事件同样与状态变更同事务写入 outbox
                eventPublisher.publish(
                    OrderEvent(
                        orderId = current.id,
                        type = "ORDER_CANCELLED",
                        occurredAt = current.updatedAt,
                        payload = mapOf("status" to current.status.name, "version" to current.version.toString()),
                    ),
                )
                current
            }
            cancelled ?: throw OrderNotFoundException(id)
        }

    suspend fun list(page: Int, size: Int, status: OrderStatus?, channel: Channel): OrderPage =
        observed("order.list", mapOf("page" to page.toString(), "size" to size.toString()), channel) {
            val offset = page * size
            val items = repository.findPage(status, offset, size)
            val total = repository.count(status)
            OrderPage(
                items = items,
                page = page,
                size = size,
                totalElements = total,
                totalPages = if (total == 0L) 0 else ((total + size - 1) / size).toInt(),
            )
        }

    /**
     * 统一观测包装：手动 span + 计时 + 成功/失败计数。
     * span 在 finally 结束，异常路径也不会泄漏；指标计时包住整个业务块。
     */
    private suspend fun <T> observed(
        spanName: String,
        attributes: Map<String, String>,
        channel: Channel,
        block: suspend () -> T,
    ): T {
        val span = tracer.nextSpan().name(spanName).apply {
            attributes.forEach { (key, value) -> tag(key, value) }
        }.start()
        return try {
            metrics.timed(channel) { block() }
        } finally {
            span.end()
        }
    }
}

timer.recordCallable { }❌ 编译失败—只能用于同步代码

D3.2.3 标签基数治理 ​

标签基数 = 每个标签取值数量的乘积。result(2) × channel(2) × region(10) × version(50) 已经是 2000 条时间序列，再叠加 userId 就是灾难。

症状根因修法
Prometheus 内存飙升标签里有 ID / URL / 异常文本标签只留枚举；ID 放 trace
查询变慢序列数过多降低维度，必要时用 exemplar
面板出现长尾标签运行时拼字符串值对象 + 编译期枚举
无法算 P99只记了 maxpublishPercentileHistogram()
  • ✅ 标签设计三问：能不能枚举？会不会超过 10 个值？删掉它会不会影响决策？
  • ⚠️ 公共标签用 management.metrics.tags.* 统一注入（本示例注入 application=backend-service），避免每个指标重复写。
  • ⚠️ 高基数需求用 exemplar 或日志，而不是新标签。

D3.3.2 HTTP / gRPC / 协程传播 ​

边界传播方式常见坑
HTTP 入站Micrometer Servlet Filter 自动解析网关剥掉 traceparent 导致断链
HTTP 出站RestTemplate/WebClient 自动注入手写 HttpURLConnection 不注入
gRPCClient/Server interceptor 注入 metadata忘记注册 interceptor，链路断在服务边界
协程Context.asContextElement()只在根协程加一次即可，子协程自动继承
ReactorHooks.enableAutomaticContextPropagation()与协程混用时统一到 OTel Context
定时任务/MQ消息头携带 traceparent，消费端恢复生产端不埋点，消费端就是新 trace

本仓库 Outbox 场景把 traceparent 保存在 outbox.trace_parent，发布器和消费端都通过 TraceContextPropagation.extract(...) 恢复父上下文，实现真正的跨消息链接（见 D3.7.1）。

kotlin
// 文件: OrderController.kt —— Web 入口：OTel Context 跨协程传播 + 鉴权后的订单生命周期接口
// 配套章节：04-jvm-backend.md 4.7 / 4.8、mastery-observability.md D3.2 / D3.5
package com.example.backend

import com.example.backend.order.CreateOrderCommand
import com.example.backend.order.Order
import com.example.backend.order.OrderStatus
import io.opentelemetry.api.trace.Span
import io.opentelemetry.context.Context
import io.opentelemetry.extension.kotlin.asContextElement
import jakarta.validation.Valid
import jakarta.validation.constraints.Max
import jakarta.validation.constraints.Min
import jakarta.validation.constraints.NotBlank
import jakarta.validation.constraints.Positive
import jakarta.validation.constraints.Size
import kotlinx.coroutines.withContext
import org.slf4j.LoggerFactory
import org.springframework.http.HttpHeaders
import org.springframework.http.HttpStatus
import org.springframework.http.ResponseEntity
import org.springframework.validation.annotation.Validated
import org.springframework.web.bind.annotation.GetMapping
import org.springframework.web.bind.annotation.PathVariable
import org.springframework.web.bind.annotation.PostMapping
import org.springframework.web.bind.annotation.RequestBody
import org.springframework.web.bind.annotation.RequestHeader
import org.springframework.web.bind.annotation.RequestMapping
import org.springframework.web.bind.annotation.RequestParam
import org.springframework.web.bind.annotation.RestController

data class CreateOrderRequest(
    @field:NotBlank @field:Size(max = 64) val customerId: String,
    @field:NotBlank @field:Size(max = 200) val itemName: String,
    @field:Positive @field:Max(1_000_000_000) val amountCents: Long,
)

data class OrderResponse(
    val id: Long,
    val name: String,                 // 兼容既有 Pact 契约：name = itemName
    val status: String,
    val amountCents: Long,
    val customerId: String,
    val version: Long,
    val createdAt: String,
    val traceId: String,
)

data class PageResponse<T>(
    val items: List<T>,
    val page: Int,
    val size: Int,
    val totalElements: Long,
    val totalPages: Int,
)

@Validated
@RestController
@RequestMapping("/api/orders")
class OrderController(private val service: OrderService) {

    private val log = LoggerFactory.getLogger(javaClass)

    /**
     * Spring MVC 会为每个请求创建 server span，但 suspend 恢复时可能换线程，
     * ThreadLocal 里的 trace 上下文会丢失。这里在挂起前抓取 OTel Context，
     * 再用 `asContextElement()` 挂到协程上下文上，保证下游 span 仍是同一个 trace。
     */
    @PostMapping
    suspend fun create(
        @RequestHeader("Idempotency-Key") @Size(min = 8, max = 128) idempotencyKey: String,
        @Valid @RequestBody request: CreateOrderRequest,
    ): ResponseEntity<OrderResponse> = traced { traceId ->
        log.info("create order start key={} customer={}", idempotencyKey, request.customerId)
        val result = service.create(
            CreateOrderCommand(
                customerId = request.customerId,
                itemName = request.itemName,
                amountCents = request.amountCents,
                idempotencyKey = idempotencyKey,
            ),
            Channel.WEB,
        )
        ResponseEntity
            .status(if (result.replayed) HttpStatus.OK else HttpStatus.CREATED)
            .header(HttpHeaders.LOCATION, "/api/orders/${result.order.id}")
            .body(result.order.toResponse(traceId))
    }

    @GetMapping("/{id}")
    suspend fun get(@PathVariable id: Long): OrderResponse = traced { traceId ->
        log.info("load order start orderId={}", id)
        service.load(id, Channel.WEB).toResponse(traceId)
    }

    @GetMapping
    suspend fun list(
        @RequestParam(defaultValue = "0") @Min(0) page: Int,
        @RequestParam(defaultValue = "20") @Min(1) @Max(100) size: Int,
        @RequestParam(required = false) status: OrderStatus?,
    ): PageResponse<OrderResponse> = traced { traceId ->
        val result = service.list(page, size, status, Channel.WEB)
        PageResponse(
            items = result.items.map { it.toResponse(traceId) },
            page = result.page,
            size = result.size,
            totalElements = result.totalElements,
            totalPages = result.totalPages,
        )
    }

    @PostMapping("/{id}/cancel")
    suspend fun cancel(@PathVariable id: Long): OrderResponse = traced { traceId ->
        log.info("cancel order start orderId={}", id)
        service.cancel(id, Channel.WEB).toResponse(traceId)
    }

    private suspend fun <T> traced(block: suspend (String) -> T): T {
        val otelContext = Context.current()
        val traceId = Span.current().spanContext.traceId
        return withContext(otelContext.asContextElement() + MdcContextElement(mapOf("traceId" to traceId))) {
            block(traceId)
        }
    }

    private fun Order.toResponse(traceId: String): OrderResponse =
        OrderResponse(
            id = id,
            name = itemName,
            status = status.name,
            amountCents = amountCents,
            customerId = customerId,
            version = version,
            createdAt = createdAt.toString(),
            traceId = traceId,
        )
}

D3.4 日志与 MDC ​

结构化日志的三条底线：

  1. 参数化：log.info("order={}", id)，不要字符串拼接。
  2. 有 traceId：MDC 里的 traceId 让日志能与链路互跳。
  3. 不打印敏感信息：手机号、身份证、token 必须脱敏。

协程换线程会让 MDC 丢失，用 ThreadContextElement 显式搬运：

kotlin
// 文件: MdcContextElement.kt —— 把 MDC 带进协程，并在切回线程时恢复
// 配套章节：04-jvm-backend.md 4.7、mastery-observability.md D3.5
package com.example.backend

import kotlinx.coroutines.ThreadContextElement
import org.slf4j.MDC
import kotlin.coroutines.CoroutineContext

/**
 * 协程会在线程间跳动，而 MDC 是 ThreadLocal：不显式传递，日志里的 traceId 就会丢。
 * 这个元素在「进入协程线程」时写入 MDC，在「离开」时恢复原值（而不是简单 clear）。
 */
class MdcContextElement(
    private val context: Map<String, String>,
) : ThreadContextElement<Map<String, String>> {

    companion object Key : CoroutineContext.Key<MdcContextElement>

    override val key: CoroutineContext.Key<*> = Key

    override fun updateThreadContext(context: CoroutineContext): Map<String, String> {
        val previous = MDC.getCopyOfContextMap() ?: emptyMap()
        this.context.forEach { (k, v) -> MDC.put(k, v) }
        return previous
    }

    override fun restoreThreadContext(context: CoroutineContext, oldState: Map<String, String>) {
        MDC.clear()
        oldState.forEach { (k, v) -> MDC.put(k, v) }
    }
}

D3.5 SLO、错误预算与燃烧率 ​

SLO 是「对内承诺」，SLI 是「实际测量」，错误预算是「允许的失败额度」。

examples/backend-service/src/main/kotlin/com/example/backend/SloBudget.kt可在线运行
kotlin
// 文件: SloBudget.kt —— SLO 目标、错误预算与燃烧率的最小实现
// 配套章节：mastery-observability.md D3.6
package com.example.backend

/**
 * @param targetSuccessRate 目标成功率，如 0.999 表示 99.9%
 * @param windowMinutes 统计窗口（如 30 天）
 */
data class Slo(
    val targetSuccessRate: Double,
    val windowMinutes: Double,
) {
    /** 窗口内允许失败的比例；0.999 → 0.001。 */
    val allowedFailureRatio: Double get() = 1.0 - targetSuccessRate

    /** 错误预算：窗口内允许失败的总时长（分钟）。 */
    fun errorBudgetMinutes(): Double = windowMinutes * allowedFailureRatio

    /** 实际失败时长占错误预算的比例；> 1 表示预算已超支。 */
    fun budgetConsumedRatio(failedMinutes: Double): Double =
        if (errorBudgetMinutes() == 0.0) Double.POSITIVE_INFINITY
        else failedMinutes / errorBudgetMinutes()

    /** 燃烧率：按当前速度会在窗口内用掉几倍预算；1 表示刚好用完。 */
    fun burnRate(failedMinutes: Double, elapsedMinutes: Double): Double {
        if (elapsedMinutes <= 0.0) return 0.0
        val projected = failedMinutes / elapsedMinutes * windowMinutes
        return budgetConsumedRatio(projected)
    }
}

D3.7.1 OTLP 导出与消息链路（D3 补全） ​

D3 交付时只有进程内 span。本次补齐两条链路：

  1. 导出：引入 opentelemetry-exporter-otlp，但只有 shop.observability.otlp-endpoint 非空时才注册 SpanExporter（Spring Boot 的 SpanExporters 自动包成 BatchSpanProcessor）。端点为空时组件不注册，启动、测试、bootRun 都不会因为缺少 collector 失败。
  2. 消息传播：业务事务内抓取 traceparent 写入 outbox.trace_parent；发布器与消费端各自 extract 还原父上下文。因此 HTTP → 协程 → Outbox relay → 消费端是同一条 trace，而不是消费端新开一条。

// 文件: observability/OtlpExporterConfiguration.kt —— 端点存在时才注册 OTLP exporter
// 与 Spring Boot 的 SpanExporters/SpanProcessors 自动装配配合：这里只提供 SpanExporter，
// BatchSpanProcessor 与 SdkTracerProvider 由 Spring Boot 自动串联。
package com.example.backend.observability

import io.opentelemetry.exporter.otlp.http.trace.OtlpHttpSpanExporter
import io.opentelemetry.sdk.trace.export.SpanExporter
import org.springframework.boot.autoconfigure.condition.ConditionalOnClass
import org.springframework.context.annotation.Bean
import org.springframework.context.annotation.Condition
import org.springframework.context.annotation.ConditionContext
import org.springframework.context.annotation.Conditional
import org.springframework.context.annotation.Configuration
import org.springframework.core.type.AnnotatedTypeMetadata
import java.time.Duration

/** 只有显式配置非空端点才启用导出；未配置时保持「进程内 span」的既有行为。 */
class OtlpEndpointConfigured : Condition {
    override fun matches(context: ConditionContext, metadata: AnnotatedTypeMetadata): Boolean =
        context.environment.getProperty("shop.observability.otlp-endpoint")
            ?.trim()
            ?.isNotEmpty() == true
}

@Configuration(proxyBeanMethods = false)
@ConditionalOnClass(OtlpHttpSpanExporter::class)
class OtlpExporterConfiguration {

    @Bean
    @Conditional(OtlpEndpointConfigured::class)
    fun otlpSpanExporter(properties: ObservabilityProperties): SpanExporter =
        OtlpHttpSpanExporter.builder()
            .setEndpoint(requireNotNull(properties.otlpEndpoint).trim())
            .setTimeout(Duration.ofSeconds(10))
            .build()
}

// 文件: observability/TraceContextPropagation.kt —— W3C traceparent 的注入与提取
// 用于 HTTP → 协程 → Outbox → 消费端的跨进程链路：事件落库时带上 traceparent，
// 发布/消费时还原 OTel Context，而不是「消费端重新开一条 trace」。
package com.example.backend.observability

import io.opentelemetry.api.trace.Span
import io.opentelemetry.api.trace.propagation.W3CTraceContextPropagator
import io.opentelemetry.context.Context
import io.opentelemetry.context.propagation.TextMapGetter
import io.opentelemetry.context.propagation.TextMapSetter

object TraceContextPropagation {

    const val TRACE_PARENT = "traceparent"

    private val propagator = W3CTraceContextPropagator.getInstance()

    private val setter = TextMapSetter<MutableMap<String, String>> { carrier, key, value ->
        carrier?.set(key, value)
    }

    private val getter = object : TextMapGetter<Map<String, String>> {
        override fun keys(carrier: Map<String, String>): Iterable<String> = carrier.keys
        override fun get(carrier: Map<String, String>?, key: String): String? = carrier?.get(key)
    }

    /**
     * 从当前 span 生成 W3C traceparent。
     * 没有有效 span（例如脱离 Web 请求的定时任务）时返回 null，不伪造上下文。
     */
    fun currentTraceParent(): String? {
        val context = Context.current()
        if (!Span.fromContext(context).spanContext.isValid) return null
        val carrier = mutableMapOf<String, String>()
        propagator.inject(context, carrier, setter)
        return carrier[TRACE_PARENT]
    }

    /** 还原上下文；缺失或非法 traceparent 退化为 root（相当于开一条新链路）。 */
    fun extract(traceParent: String?): Context {
        if (traceParent.isNullOrBlank()) return Context.root()
        return propagator.extract(Context.root(), mapOf(TRACE_PARENT to traceParent), getter)
    }
}
